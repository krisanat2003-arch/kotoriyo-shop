เปิด Apps Script และวางโค้ด
สร้าง API ที่เว็บจะใช้ดึงและบันทึกข้อมูล

ใน Google Sheet ที่เพิ่งสร้าง คลิกเมนู Extensions → Apps Script

ลบโค้ดเดิมทิ้งทั้งหมด แล้ว copy โค้ดด้านล่างนี้ไปวางแทน:

Google Apps Script 
// Koto Riyo Shop — Google Apps Script API
// วางโค้ดนี้ใน Extensions → Apps Script แล้ว Deploy

function doGet(e) {
  const action = e.parameter.action || 'products';
  let result;

  if (action === 'products') {
    result = getProducts();
  } else if (action === 'saveOrder') {
    result = saveOrder(e.parameter);
  } else {
    result = { error: 'Unknown action' };
  }

  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function getProducts() {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet()
      .getSheetByName('Products');
    if (!sheet) return { error: 'ไม่พบ sheet ชื่อ Products' };

    const data = sheet.getDataRange().getValues();
    if (data.length < 2) return [];

    const headers = data[0];
    const products = data.slice(1)
      .filter(row => row[0] !== '')
      .map(row => {
        const obj = {};
        headers.forEach((h, i) => { obj[h] = row[i]; });
        return obj;
      })
      .filter(p => p.inStock === true || p.inStock === 'TRUE');

    return products;
  } catch(err) {
    return { error: err.toString() };
  }
}

function saveOrder(params) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet()
      .getSheetByName('Orders');
    if (!sheet) return { error: 'ไม่พบ sheet ชื่อ Orders' };

    const orderId = 'KR-' + new Date().getTime().toString().slice(-6);
    sheet.appendRow([
      orderId,
      new Date(),
      params.name   || '',
      params.email  || '',
      params.phone  || '',
      params.address|| '',
      params.items  || '',
      params.total  || 0,
      params.payment|| '',
      'รอดำเนินการ'
    ]);

    return { success: true, orderId: orderId };
  } catch(err) {
    return { error: err.toString() };
  }
}

หลังวางโค้ดแล้ว กด Save (Ctrl+S)

3
Deploy เป็น Web App
เปิดให้เว็บของเราเรียกใช้งาน API ได้

ใน Apps Script คลิกปุ่ม Deploy → New deployment (มุมขวาบน)

ตั้งค่าดังนี้:

①
Type
Web app
②
Execute as
Me (อีเมลของคุณ)
③
Who has access
Anyone (สำคัญมาก!)

กด Deploy → อนุญาต permissions → คัดลอก Web app URL ที่ได้มา

⚠️ URL จะหน้าตาแบบนี้: https://script.google.com/macros/s/XXXXXXXX/exec
🔗 ขั้นตอนที่ 4 — วาง Web App URL ที่นี่
✅ บันทึก URL เรียบร้อยแล้ว! เว็บจะดึงสินค้าจาก Google Sheets ทันที
🚀 ไปที่หน้าร้านค้า
